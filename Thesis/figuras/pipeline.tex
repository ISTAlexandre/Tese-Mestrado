\begin{tikzpicture}[

  box/.style={draw, rounded corners=2pt, align=center, text width=3.8cm,

% ---- colunas: A = EEEC, B = dados, C = simulacao --------------------------
\def\xA{0} \def\xB{5.3} \def\xC{10.6}

% ---- entradas ---------------------------------------------------------------
\node[io] (inD) at (\xB,0)   {\textbf{JetHT, Run 2016G}\\ MiniAOD (collision data)};
\node[io] (inS) at (\xC,0)   {\textbf{QCD multijet, Pythia 8 CP5}\\ MiniAODSIM (simulation)};

% ---- ambiente CMS -----------------------------------------------------------
\node[proc] (exD) at (\xB,-2.0) {\textbf{Extraction and selection}\\

      $p_{\mathrm{T}}>700$~GeV, $|\eta|<1.7$, dijet};
\node[proc] (exS) at (\xC,-2.0) {\textbf{Extraction and selection}\\
      same jet selection\\ + final-state partons};
\node[io]   (ntD) at (\xB,-4.0) {\textbf{Jet ntuple}\\ jets and constituents};
\node[io]   (ntS) at (\xC,-4.0) {\textbf{Jet ntuple}\\ jets, constituents, partons};

% ---- ambiente local ---------------------------------------------------------
\node[proc] (ee)  at (\xA,-6.4) {\textbf{Energy correlator}\\

\node[proc] (dcD) at (\xB,-6.4) {\textbf{Declustering}\\
      C/A reclustering, Lund\\ declustering, Soft Drop\\
      $\to$ $\Delta\psi_{12}$, subjet shapes};
\node[proc] (dcS) at (\xC,-6.4) {\textbf{Declustering}\\ identical to data\\
      $\to$ $\Delta\psi_{12}$, subjet shapes};
\node[proc] (lbS) at (\xC,-8.7) {\textbf{Channel labelling}\\
      partons: anti-$k_{\mathrm{T}}$, then C/A + IFN\\
      gen--reco jet matching\\ $\to$ unmatched, $q\bar q$, $qg$, $gg$};
\node[ml]   (trS) at (\xC,-11.0) {\textbf{DNN training}\\
      stage 1: matched vs unmatched\\ stage 2: $g\to q\bar q$ vs rest};
\node[ml]   (apD) at (\xB,-11.0) {\textbf{DNN application}\\
      matched and $g\to q\bar q$\\ scores per jet};
\node[res]  (eeR) at (\xA,-13.3) {\textbf{EEEC}\\ $\psi$ distribution};
\node[res]  (ms)  at (\xB,-13.3) {\textbf{$\Delta\psi_{12}$ measurement}\\
      $\langle\cos 2\Delta\psi_{12}\rangle$ vs\\ $g\to q\bar q$ purity};

% ---- setas ------------------------------------------------------------------
